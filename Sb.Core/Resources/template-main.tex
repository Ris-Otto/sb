% The body font is 12pt on 14.4pt leading. A \fontsize in the preamble is undone at \begin{document}
% (LaTeX selects \normalsize then, which is 10pt in the book class), so \normalsize itself is
% redefined; headings, which use \normalsize, then match the text.

\makeatother
